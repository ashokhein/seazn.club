## Design notes carried from the audit (read before starting)

- **The resolver is already rung-agnostic.** `org_has_feature` (SQL, `db/migration/deltas/V328__event_pass_lifecycle_lock.sql:79-90`) joins `plan_entitlements pe on pe.plan_key = cp.pass_key`, and `resolveFromDb` (TS, `apps/web/src/lib/entitlements.ts:309-317`) does the identical join in JS. Neither hardcodes `'event_pass'`. The lock (`isPassLocked` / the SQL `not (c.status in ('archived','completed') or ...)` arm) is keyed on the competition's status/`ends_on`, never on `pass_key`. **This means Task 1's migration is the only thing standing between "no L rung" and "L rung fully works everywhere the resolver is read"** — no entitlements code changes at all.
- **`competition_passes.pass_key` already exists** as a column (`db/migration/deltas/V271__competition_passes.sql:9`: `pass_key text not null default 'event_pass' references plans(key)`), it is just never written as anything but the default today.
- **The M rung's plan key is literally `event_pass`** (not `event_pass_m`) — seeded in `db/migration/deltas/V270__pricing_v3_matrix.sql:21`. The L rung's key is `event_pass_l`.
- **`PASS_CREDIT_GRANT = 25`** (`apps/web/src/lib/pricing-cards.ts:45`) is a flat constant `recordPassGrant` always applies (`apps/web/src/lib/credits.ts:498-520`, called via `apps/web/src/lib/billing.ts:807-810`) — it does not read `pass_key` at all. Per the owner decision, L grants the same 25; **do not parametrize this constant.**
- **Both `reconcilePassCheckout` (`apps/web/src/lib/billing.ts:916-968`) and the webhook (`apps/web/src/server/usecases/billing-events.ts:191-227`) currently gate on `session.metadata?.pass_key === "event_pass"`.** This is the ONE place that must widen to recognise `event_pass_l`, with a safe fallback to `event_pass` for any session whose `pass_key` is present-but-unrecognised (defensive; every real session has always stamped `pass_key: "event_pass"` since v3/07, so this is a robustness net, not a live migration path). A session with **no** `pass_key` at all must still be refused — that is how the pass branch is told apart from subscription-checkout sessions, which never set it (`apps/web/src/lib/billing.ts:165-168`).
- **Money-safety machinery (refund clawback, pass-to-Pro subscription credit) never reads `pass_key` at all** — `revokePassForRefundedCharge` (`apps/web/src/lib/billing.ts:872-883`) keys purely on `stripe_payment_intent`, and `server/usecases/pass-credit.ts` (the pass-to-Pro **customer balance** credit, a different mechanism from the AI-credit wallet grant) never references `pass_key` either. Regression tests in Task 3 exist to **prove** this, not to change it.
- **The purchase UI must not open a picker as a separate step before Stripe** — `apps/web/e2e/event-pass.spec.ts:325` and `:922` click `[data-pass-buy]` and immediately wait for the Stripe iframe. The M/L choice must therefore be **inline** in the ticket stub (radios pre-selecting M), with `[data-pass-buy]` staying the single button that starts checkout for whichever rung is currently selected. This preserves the entire e2e money-path suite unchanged (verified: no e2e assertion pins button text, only `[data-pass-buy]`, `[data-pass-ticket]` containing `"$29"`, and `[data-pass-active]` — all still true with M pre-selected by default).

  ```sql
  -- carries, L carries identically, except the two overrides below.
  insert into plans (key, name, is_public) values ('event_pass_l', 'Event Pass L', false)
  on conflict (key) do nothing;

  insert into plan_entitlements (plan_key, feature_key, bool_value, int_value)

  from plan_entitlements
  where plan_key = 'event_pass'
  on conflict (plan_key, feature_key) do update
    set bool_value = excluded.bool_value, int_value = excluded.int_value;

  insert into plan_entitlements (plan_key, feature_key, bool_value, int_value) values
  ```

  ```tsx
  export function PassCheckoutSheet({

  }: {
    clientSecret: string;
    onClose: () => void;
  }) {
    return (
      <Modal title="Complete your purchase" size="lg" onClose={onClose}>

  }

  /**
   * In-page Event Pass purchase (v17 #294 / spec A7): an inline M/L radio
   * picker — M pre-selected — feeding the SAME single buy button the pre-#294
   * UI had. Deliberately NOT a separate "choose, then pay" modal step
   * (compare buy-credits.tsx): apps/web/e2e/event-pass.spec.ts clicks
   * `[data-pass-buy]` and waits directly for the Stripe iframe, so the button
   * must keep starting checkout on the FIRST click, for whichever rung is
   * currently selected. No "best value" badge — the owner decided #294 ships
   * without one, unlike the credit-pack ladder.
   */
  export function PassUpgradeButton({
    competitionId,
    options,
    currency,
    dict,
    label,
  }: {
    competitionId: string;
    /** Exactly two rungs, event_pass first — lib/pass-ladder.ts's shape. */
    options: PassRungOption[];
    currency: Currency;
    dict: Dict;
    /** "Buy the pass" — the trigger's base copy; the rung name is appended. */
    label: string;
  }) {
    const [selected, setSelected] = useState<PassKey>("event_pass");
    const [clientSecret, setClientSecret] = useState<string | null>(null);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState<string | null>(null);

    async function start() {
      setError(null);
      setLoading(true);
      track(EVENTS.CHECKOUT_STARTED, { plan_key: selected });
      const result = await fetchPassCheckoutClientSecret(competitionId, selected);
      setLoading(false);
      if (result.ok) setClientSecret(result.clientSecret);
      else setError(result.error);
    }

    if (clientSecret) {
      return <PassCheckoutSheet clientSecret={clientSecret} onClose={() => setClientSecret(null)} />;
    }

    const rungLabel = (key: PassKey) =>
      t(dict, key === "event_pass" ? "upgrade.ladder.rungM" : "upgrade.ladder.rungL");

    return (
      <div>
        <fieldset className="mb-3 space-y-1.5 text-left">
          <legend className="sr-only">{t(dict, "upgrade.ladder.title")}</legend>
          {options.map((option) => {
            const active = option.key === selected;
            const entrantsText =
              option.entrants === null
                ? t(dict, "upgrade.ladder.entrantsUnlimited")
                : t(dict, "upgrade.ladder.entrantsCapped", { count: option.entrants });
            const divisionsText =
              option.divisions === null
                ? t(dict, "upgrade.ladder.divisionsUnlimited")
                : t(dict, "upgrade.ladder.divisionsCapped", { count: option.divisions });
            return (
              <label
                key={option.key}
                data-pass-rung={option.key}
                className={`flex cursor-pointer items-start gap-2 rounded-lg border px-2.5 py-1.5 text-xs transition ${
                  active ? "border-lime-300 bg-white/10" : "border-white/15 hover:border-white/25"
                }`}
              >
                <input
                  type="radio"
                  name="pass-rung"
                  value={option.key}
                  checked={active}
                  onChange={() => setSelected(option.key)}
                  className="mt-0.5 h-3.5 w-3.5 accent-lime-400"
                />
                <span className="flex flex-1 flex-col">
                  <span className="font-semibold text-white">
                    {rungLabel(option.key)} {formatMinor(option.amountMinor, currency)}
                  </span>
                  <span className="text-white/60">
                    {divisionsText} · {entrantsText}
                  </span>
                </span>
              </label>
            );
          })}
        </fieldset>
        <p className="mb-2 text-[0.6875rem] text-white/50">{t(dict, "upgrade.ladder.note")}</p>
        <button
          type="button"
          onClick={start}
          disabled={loading}
          className="btn btn-primary px-5 py-2.5"
          data-pass-buy
        >
          {loading
            ? t(dict, "upgrade.ladder.preparing")
            : `${label} — ${rungLabel(selected)}`}
        </button>
        {error && <p className="mt-2 text-sm text-red-600">{error}</p>}
      </div>
    );
  }
  ```

  ```ts
  /**
   * v17 #294 — the L rung. Mirrors passGrantsSuite's entrants/divisions PAIR
   * shape (passed vs sibling) so a leak or a wrong ceiling fails for the right
   * reason. The credit grant and lock/refund machinery are already proven

  async function passGrantsLSuite(): Promise<void> {
    const genericDiv = {
      sport_key: "generic",
      variant_key: "score",
      config: { points: { w: 3, d: 1, l: 0 }, progressScore: false },
    };
    const featureKey = (r: V1Res) => (r.json.error as { feature_key?: string } | undefined)?.feature_key;
    const entrants = (n: number, from: number, label: string) =>
      Array.from({ length: n }, (_, i) => ({ kind: "individual", display_name: `${label}${from + i}`, seed: from + i }));

    const mkComp = async (name: string) =>
      v1data<{ id: string; slug: string }>(
        await v1(s, "/api/v1/competitions", "POST", { name: `${name} ${tag}`, visibility: "unlisted" }),
      );
    const mkDiv = async (compId: string, name: string) =>
      v1data<{ id: string }>(
        await v1(s, `/api/v1/competitions/${compId}/divisions`, "POST", { name, ...genericDiv }),
      );

    const passComp = await mkComp("Grants L Passed");
    const plainComp = await mkComp("Grants L Plain");
    await grantPass(orgId, passComp.id, "event_pass_l");
    check(
      "pass grants L: fixture built — one community org, one L-passed competition",
      !!passComp.id && !!plainComp.id,

      seat200.status === 201,

    let passDivisionsOk = true;

      ...genericDiv,
  ```
